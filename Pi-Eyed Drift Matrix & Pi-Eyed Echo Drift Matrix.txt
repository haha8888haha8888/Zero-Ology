\Omega_{\text{drift}} = (A-B)(1+f_A) + (B-C)(1+f_B) - (A-C)(1+f_A)
Expanding A-C = (A-B) + (B-C) and canceling the (A-B) terms yields:


\Omega_{\text{drift}} = \boxed{(B-C)(f_B - f_A)}

class EchoDriftMatrixNode:
    """Educational Pi-Eyed Echo Drift Matrix Node with local conservation,
    shadow-to-explicit echo feedback dynamics, and state tracking."""
    def __init__(self, explicit_val: float, retention_alpha: float, echo_beta: float, label: str = "Node"):
        self.label = label
        self.explicit = explicit_val
        self.shadow = 0.0
        self.alpha = retention_alpha  # 0 < alpha <= 1
        self.beta = echo_beta          # 0 <= beta <= 1
        self.total_ledger = explicit_val  # Strict conservation baseline T


    def step_dynamics(self):
        """Executes local echo feedback update preserving E + S == Constant."""
        e_curr = self.explicit
        s_curr = self.shadow
        self.explicit = self.alpha * e_curr + self.beta * s_curr
        self.shadow = (1.0 - self.beta) * s_curr + (1.0 - self.alpha) * e_curr


    def verify_conservation(self, tolerance: float = 1e-12) -> bool:
        return abs((self.explicit + self.shadow) - self.total_ledger) < tolerance




def calculate_directed_drift(node_i, node_j) -> float:
    """Computes directed drift D(A,B) using the active initiator's forgetfulness."""
    raw_gap = abs(node_i.explicit - node_j.explicit)
    return raw_gap * (1.0 + (1.0 - node_i.alpha))
