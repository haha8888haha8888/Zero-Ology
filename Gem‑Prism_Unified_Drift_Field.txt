@dataclass
class EchoDriftNode:
    E: float          # explicit
    S: float          # shadow
    alpha: float      # retention
    beta: float       # echo


    def step(self):
        """One step of Echo Drift dynamics."""
        E_next = self.alpha * self.E + self.beta * self.S
        S_next = (1.0 - self.alpha) * self.E + (1.0 - self.beta) * self.S
        self.E, self.S = E_next, S_next


    @property
    def T(self) -> float:
        """Total ledger (conserved quantity)."""
        return self.E + self.S


    def verify_conservation(self, T0: float, tol: float = 1e-12) -> bool:
        return abs(self.T - T0) < tol


    def lambda_eigs(self) -> Tuple[float, float]:
        """Return eigenvalues λ1, λ2 of the local transition matrix."""
        # P = [[alpha, beta], [1-alpha, 1-beta]]
        # eigenvalues: 1 and alpha - beta
        return 1.0, self.alpha - self.beta

def forgetfulness(alpha: float) -> float:
    return 1.0 - alpha




def directed_drift(E_i: float, E_j: float, alpha_i: float) -> float:
    f_i = forgetfulness(alpha_i)
    return abs(E_i - E_j) * (1.0 + f_i)




def drift_residual_numeric(EA: float, EB: float, EC: float,
                           alphaA: float, alphaB: float) -> float:
    """Compute Ω(A,B,C) numerically from D."""
    d_ab = directed_drift(EA, EB, alphaA)
    d_bc = directed_drift(EB, EC, alphaB)
    d_ac = directed_drift(EA, EC, alphaA)
    return d_ab + d_bc - d_ac




def drift_residual_analytic(EA: float, EB: float, EC: float,
                            alphaA: float, alphaB: float) -> float:
    """Analytic ordered-case Ω = (EB - EC)(fB - fA)."""
    fA = forgetfulness(alphaA)
    fB = forgetfulness(alphaB)
    return (EB - EC) * (fB - fA)
